UPDATED ROUTER LOGIC function showView(view) { const port = document.getElementById('view-port'); document.getElementById('view-title').innerText = view.toUpperCase(); document.querySelectorAll('.nav-item').forEach(i => i.classList.remove('active')); document.getElementById('nav-' + view).classList.add('active'); if (view === 'home') { port.innerHTML = `

RCM Master Command

Database Status: Connected

Total Records: ${masterData.length}

`; } else if (view === 'search') { // FILTER: Remove Radiology/CPT codes from this view const denialOnly = masterData.filter(d => !d.cat.toLowerCase().includes('rad') && !d.code.startsWith('7') ); port.innerHTML = `

Denial Management (CARC/RARC)

${renderRows(denialOnly)}
CodeCategoryDescriptionResolution
`; } else if (view === 'radiology') { // FILTER: Only show Radiology/CPT codes here const radOnly = masterData.filter(d => d.cat.toLowerCase().includes('rad') || d.code.startsWith('7') ); port.innerHTML = `

Radiology & Imaging CPT Matrix

${renderRows(radOnly)}
CPTCategoryDescriptionResolution Logic
`; } } // 2. HELPER TO RENDER ROWS function renderRows(data) { if (data.length === 0) return 'No records found.'; return data.map(d => { // Apply different badge colors based on category let colorClass = 'tech'; if (d.cat.toLowerCase().includes('elig')) colorClass = 'elig'; if (d.cat.toLowerCase().includes('rad')) colorClass = 'rad'; return ` ${d.code} ${d.cat} ${d.desc} ${d.res} `; }).join(''); }
